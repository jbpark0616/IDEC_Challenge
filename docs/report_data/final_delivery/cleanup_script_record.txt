$ErrorActionPreference='Stop'
$root=[IO.Path]::GetFullPath((Join-Path $PSScriptRoot '..'))
Set-Location $root
$targets=@(
 'docs/papers','docs/수상작','docs/cleanup_20261030',
 'docs/IDEC_중간보고서.docx','docs/IDEC_중간보고서.pdf',
 'docs/gemm_centric_winograd_architecture.md','docs/temporal_sparse_multicore_winograd.md',
 'docs/winograd_full_bank_engine.md','docs/winograd_f2x5_overhead.md',
 'scripts/apply_free_topic_cleanup.ps1','scripts/audit_disk_space.py','scripts/audit_project_split.py',
 'scripts/cleanup_designated_generated.ps1','scripts/cleanup_final_delivery.ps1',
 'scripts/finalize_free_topic_cleanup_plan.py','scripts/prepare_free_topic_cleanup.py',
 'free_topic_vgg/verification/feed_hwc4','free_topic_vgg/verification/feed_hwc8',
 'free_topic_vgg/verification/raster_restore',
 'free_topic_vgg/verification/bypass_profile/cluster_immediate.py',
 'free_topic_vgg/verification/bypass_profile/feed_address_audit.py',
 'free_topic_vgg/verification/bypass_profile/report_cluster_probe.py'
)
foreach($name in @('cluster_bottleneck','cluster_immediate','feed8_bypass_audit','feed8_resident_more','feed_group_sweep','feed_hwc8','raster_restore','raster_restore_immediate')){
 $targets+="free_topic_vgg/evidence/$name"
 $targets+="free_topic_vgg/evidence/${name}_run.log"
}
# Downloaded paper copies are not design or report deliverables.
$targets+=@(Get-ChildItem docs -File -Filter '*.pdf' | Where-Object {$_.Name -notmatch '^(IDEC_|2026 KNU)' -and $_.Name -notmatch '표지'} | ForEach-Object {$_.FullName.Substring($root.Length+1)})
$protected=@(Get-ChildItem verilog,free_topic_vgg/rtl,submission/IDEC2026 -Recurse -File)+@(Get-ChildItem docs -File | Where-Object {$_.Name -match '제출용.pdf$|^IDEC_지정주제_결과보고서.docx$|^IDEC_자유주제_결과보고서_검토수정본.docx$'})
$before=@{};foreach($f in $protected){$before[$f.FullName]=(Get-FileHash -LiteralPath $f.FullName).Hash}
$manifest=@()
foreach($rel in $targets){
 $p=[IO.Path]::GetFullPath((Join-Path $root $rel))
 if(-not $p.StartsWith($root+[IO.Path]::DirectorySeparatorChar)){throw "Outside workspace: $p"}
 if(Test-Path -LiteralPath $p){
  $files=if((Get-Item -LiteralPath $p).PSIsContainer){@(Get-ChildItem -LiteralPath $p -Recurse -File)}else{@(Get-Item -LiteralPath $p)}
  foreach($f in $files){$manifest+=[pscustomobject]@{path=$f.FullName.Substring($root.Length+1);bytes=$f.Length;sha256=(Get-FileHash -LiteralPath $f.FullName).Hash}}
 }
}
$manifest|ConvertTo-Json -Depth 4|Set-Content -Encoding UTF8 docs/report_data/final_delivery/final_only_removed.json
foreach($rel in $targets){
 $p=[IO.Path]::GetFullPath((Join-Path $root $rel))
 if(-not $p.StartsWith($root+[IO.Path]::DirectorySeparatorChar)){throw $p}
 if(Test-Path -LiteralPath $p){Remove-Item -LiteralPath $p -Recurse -Force}
}
foreach($p in $before.Keys){if((Get-FileHash -LiteralPath $p).Hash -ne $before[$p]){throw "Protected file changed: $p"}}
Write-Output "PASS final RTL, submission and report hashes unchanged; removed $($manifest.Count) files."
